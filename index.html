Mã Google Apps Script (Khớp 8 cột Chấm PSA)

Dán mã này vào Tiện ích mở rộng > Apps Script trong file Google Sheets Chấm PSA, sau đó chọn Triển khai > Ứng dụng web (Bất kỳ ai).

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0];
    var payload = JSON.parse(e.postData.contents);
    var rows = payload.rows || [];

    var formattedRows = rows.map(function(r) {
      var goiCham = String(r[4] || "").replace(/[^0-9]/g, "") || r[4];
      return [
        r[2] || "", // A: Khách hàng
        r[0] || "", // B: Sản phẩm
        r[1] || 1,  // C: Số lượng
        r[3] || "", // D: Ngày giờ
        goiCham,    // E: Gói chấm/ngày
        r[5] || 0,  // F: Giá tiền
        r[7] || 0,  // G: phí
        r[6] || ""  // H: Tình trạng đơn hàng
      ];
    });

    if (formattedRows.length > 0) {
      var lastRow = sheet.getLastRow();
      if (lastRow > 2) {
        sheet.getRange(3, 1, lastRow - 2, 8).clearContent();
      }
      sheet.getRange(3, 1, formattedRows.length, 8).setValues(formattedRows);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: formattedRows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
